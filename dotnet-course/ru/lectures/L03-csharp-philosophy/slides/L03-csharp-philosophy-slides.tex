\note{Левый код не плохой, он честный и рабочий -- так писали на C, и так
  пишут те, кто пришёл из C. Но чтобы понять его, надо исполнить его в
  голове. Правый читается сверху вниз как постановка задачи. Изменить
  задачу: ещё одно условие -- ещё один Where, другой порядок -- другой
  OrderBy. Слева то же изменение -- правка в трёх местах. LINQ подробно --
  Л5; сейчас достаточно читать названия методов.}


\begin{frame}[fragile]{Группировка и агрегация}
  \slidetop
  \begin{minipage}[t]{0.555\textwidth}
    \colhead{alt}{императивно}\par\vspace{0.5em}
\begin{lstlisting}[style=csharp,basicstyle=\ttfamily\fontsize{6.6}{7.6}\selectfont]
var totals = new Dictionary<int, decimal>();
foreach (var o in orders) {
    if (o.IsCancelled) {
        continue;
    }
    if (totals.ContainsKey(o.CustomerId)) {
        totals[o.CustomerId] += o.Amount;
    } else {
        totals[o.CustomerId] = o.Amount;
    }
}
var top = new List<KeyValuePair<int, decimal>>(totals);
top.Sort((a, b) => b.Value.CompareTo(a.Value));
if (top.Count > 3) {
    top.RemoveRange(3, top.Count - 3);
}
\end{lstlisting}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.42\textwidth}
    \colhead{accent}{декларативно}\par\vspace{0.5em}
\begin{lstlisting}[style=csharp,basicstyle=\ttfamily\scriptsize]
var top = orders
    .Where(o => !o.IsCancelled)
    .GroupBy(o => o.CustomerId)
    .Select(g => (Customer: g.Key,
        Total: g.Sum(o => o.Amount)))
    .OrderByDescending(x => x.Total)
    .Take(3)
    .ToList();
\end{lstlisting}
    \vspace{0.9em}
    \aside{Три клиента с самой большой выручкой без отменённых заказов.}\par
    \vspace{0.7em}
    \aside{Справа у результата \hi{имена полей}, слева --- \lo{Key и Value}, и читателю остаётся помнить, что из них что.}\par
  \end{minipage}
\end{frame}
\note{Слева вспомогательный словарь, ветка <<есть ключ / нет ключа>>,
  копирование в список, сортировка с ручным компаратором и обрезка. Каждая
  из этих строк -- механика, а не задача. Справа каждая строка -- шаг
  задачи. Кортеж с именами полей -- ещё один способ рассказать компилятору
  и читателю больше, об этом весь третий раздел.}

\begin{frame}{Что изменилось}
  \slidetop
  \claim{Декларативный код говорит, \hi{что} нужно получить, а не \lo{как}
    двигать индексы.}

  \begin{tabular}{@{}p{4.3cm}p{8.9cm}@{}}
    \textbf{\hi{шаг --- это слово}} & \texttt{Where}, \texttt{OrderBy},
      \texttt{Take}, \texttt{GroupBy}: читается как постановка задачи \\[0.5em]
    \textbf{\hi{нет механики}} & ни индексов, ни флагов, ни временных
      массивов --- негде ошибиться на единицу \\[0.5em]
    \textbf{\hi{сверху вниз}} & каждая строка --- один шаг, порядок строк
      --- порядок шагов \\[0.5em]
    \textbf{\hi{дёшево менять}} & новое условие --- ещё один \texttt{Where},
      а не правка в трёх местах \\
  \end{tabular}

  \slidegap
  \aside{Цена справа --- объект-перечислитель и вызов делегата на элемент:
    \lo{наносекунды}. Один запрос к базе --- \hi{миллисекунды}. Слева
    <<быстрее>> на величину, которую сервис не заметит, а читать его будут
    годами.}
\end{frame}
\note{Это и есть насмотренность: когда видишь цикл с флагом, в голове
  всплывает Where; когда видишь словарь с ContainsKey и суммой -- GroupBy.
  Декларативная запись -- не модная, а дешёвая: дешевле прочитать, дешевле
  изменить, дешевле найти ошибку. Для многих операций над массивами LINQ в
  новых версиях ещё и идёт коротким путём -- проверяет, не массив ли это.}

\begin{frame}{Из чего строятся типы}

  массива.}

\begin{frame}[fragile]{Свой обобщённый метод}

Console.WriteLine(Max(numbers));   // 9
Console.WriteLine(Max(names));     // c

static T Max<T>(IReadOnlyList<T> items) where T : IComparable<T>
{
    var best = items[0];
    for (var i = 1; i < items.Count; i++)
    {
        if (items[i].CompareTo(best) > 0)
            best = items[i];
    }
    return best;
}

  \aside{\texttt{\hi{T}} компилятор выводит при вызове. \texttt{\hi{where}}
    --- ограничение: без него \texttt{CompareTo} не скомпилируется, а тип без
    сравнения \lo{не пройдёт на вызове}.}
\end{frame}
\note{Ограничения: интерфейс или базовый класс, class или struct, new(),
  notnull. Смысл один: рассказать компилятору о T ровно столько, сколько
  нужно телу.}
