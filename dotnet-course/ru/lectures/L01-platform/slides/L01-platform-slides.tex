\begin{frame}{Четыре аббревиатуры}
  \slidetop
  \claim{Расшифруем.}

  \begin{tabular}{@{}p{1.9cm}p{9.3cm}@{}}
    \textbf{\hi{CLR}} & Common Language Runtime --- \hi{исполняет}: загружает
                        код, компилирует, управляет памятью \\[0.7em]
    \textbf{\hi{CTS}} & Common Type System --- \hi{определяет, что такое тип};
                        поэтому класс из F\# виден из C\# \\[0.7em]
    \textbf{\hi{CLS}} & Common Language Specification --- \hi{договор}
                        о совместимости между языками \\[0.7em]
    \textbf{\hi{BCL}} & Base Class Library --- \hi{базовый набор}
                        компонентов, включая примитивы \\
  \end{tabular}
\end{frame}
\note{Момент запомнить. CLS на практике почти не всплывает -- одной фразой.}

\begin{frame}{SDK и Runtime: что ставить}
  \slidetop
  \claim{\hi{SDK} себе, \lo{Runtime} туда, где живёт приложение.}

  \versus{SDK --- ставит разработчик}{%
    \begin{cbul}{accent}
      \item компилятор
      \item \texttt{dotnet} CLI и шаблоны проектов
      \item MSBuild
      \item рантайм внутри --- чтобы запускать у себя
    \end{cbul}}{Runtime --- стоит на сервере}{%
    \begin{cbul}{alt}
      \item всё, что нужно, чтобы запустить готовое
      \item компилятора нет
      \item шаблонов проектов нет
      \item заметно меньше по размеру
    \end{cbul}}
\end{frame}
\note{Что стоит на сервере -- вы обязаны понимать. Модели поставки -- большой
  кусок следующей лекции.}

\section{2. Путь кода}

\begin{frame}[plain]
  \sectionopener{2}{Путь кода}{Исходник $\rightarrow$ IL $\rightarrow$ машинный
    код: два компилятора, два этапа и всё, что из этого следует.}
\end{frame}
\note{Бюджет: 20 минут. Главный раздел лекции.}

\begin{frame}[plain]
  \vote{.NET компилируется или интерпретируется?}
\end{frame}
\note{Голосование в три руки. Правы все трое -- но пойдём по порядку.}

\begin{frame}[fragile]{Что на самом деле лежит в dll}
  \slidetop
  \begin{minipage}[t]{0.36\textwidth}
    \colhead{accent}{исходник}\par\vspace{0.7em}
\begin{lstlisting}[style=csharp]
static int Add(int a, int b)
    => a + b;
\end{lstlisting}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.60\textwidth}
    \colhead{alt}{то же самое в IL}\par\vspace{0.7em}
\begin{lstlisting}[style=term]
.method static int32 Add(int32 a, int32 b) cil managed
{
    .maxstack 8
    IL_0000: ldarg.0
    IL_0001: ldarg.1
    IL_0002: add
    IL_0003: ret
}
\end{lstlisting}
  \end{minipage}

  \vspace{1.0em}
  \aside{\hi{Ни слова о процессоре} \textperiodcentered{} \hi{сохранены имена
    и типы} --- это метаданные \textperiodcentered{} читается глазами,
    поэтому \lo{секретов там быть не должно}.}
\end{frame}
\note{Не учить IL. Увидеть три вещи. На метаданных держится половина
  платформы: отладчик, рефлексия, сериализация.}

\begin{frame}{Компилируется дважды}
  \slidetop
  \claim{Спор <<компилируется или интерпретируется>> бессмысленный.
    Правильный ответ --- \hi{компилируется дважды}.}

  \begin{center}
    \begin{tikzpicture}[font=\small,
      box/.style={draw=hairline,line width=0.8pt,fill=codebg,
                  minimum height=1.0cm,align=center,inner sep=0.5em},
      ar/.style={-{Straight Barb[length=1.8mm]},ink,line width=0.7pt}]
      \node[box] (src) {Program.cs};
      \node[box,right=2.1cm of src] (il) {dll с IL};
      \node[box,right=2.1cm of il] (mc) {машинный код\\в памяти};
      \draw[ar] (src) -- node[above,font=\scriptsize\bfseries,text=accent]{Roslyn} (il);
      \draw[ar] (il) -- node[above,font=\scriptsize\bfseries,text=alt]{JIT} (mc);
      \node[muted,font=\scriptsize,below=0.6em,align=center] at (src.south)
        {ваша машина,\\один раз};
      \node[muted,font=\scriptsize,below=0.6em,align=center] at (mc.south)
        {машина пользователя,\\при первом вызове метода};
    \end{tikzpicture}
  \end{center}

  \slidegap
  \aside{Артефакт на диске \hi{один}: dll с IL. Машинный код рождается заново
    на каждом запуске и умирает вместе с процессом.}
\end{frame}
\note{Те, кто голосовал за <<интерпретируется>>, почувствовали правильно, что
  что-то происходит в рантайме.}

\begin{frame}{Чем мы за это платим}
  \slidetop
  \claim{Два больших плюса и один минус.}

  \begin{itemize}\itemsep0.9em
    \item[\textcolor{accent}{\textbf{+}}] Артефакт \hi{не зависит от
          платформы}: одна dll работает везде, где есть рантайм.
    \item[\textcolor{accent}{\textbf{+}}] JIT знает \hi{конкретную машину}
          и её набор инструкций --- то, чего не знает компилятор C++ на вашем
          ноутбуке.
    \item[\textcolor{alt}{\textbf{--}}] \lo{Прогрев на старте}: каждый метод
          при первом вызове надо скомпилировать.
  \end{itemize}
\end{frame}
\note{Здесь я упрощаю -- дотнетчики, поправьте: JIT не <<подстраивается под
  всё>>, речь про набор инструкций. И у C++ есть -march=native.}

\begin{frame}{Многоуровневая компиляция}
  \slidetop
  \claim{Рантайм не выбирает между <<быстро и плохо>> и <<медленно и
    хорошо>>. Он делает \hi{и то и другое}, по очереди.}

  \begin{center}
    \begin{tikzpicture}[font=\small,
      box/.style={draw=hairline,line width=0.8pt,fill=codebg,
                  minimum height=1.1cm,minimum width=3.7cm,align=center,
                  inner sep=0.5em},
      ar/.style={-{Straight Barb[length=1.8mm]},ink,line width=0.7pt}]
      \node[box] (t0) {уровень 0\\быстро и грубо};
      \node[box,right=5.0cm of t0] (t1) {уровень 1\\полная оптимизация};
      \draw[ar] (t0) -- (t1);
      \node[accent,font=\scriptsize\bfseries,above=0.35em]
        at ($(t0.east)!0.5!(t1.west)$) {счётчик вызовов};
      \node[muted,font=\scriptsize,below=0.35em]
        at ($(t0.east)!0.5!(t1.west)$) {в фоне, с подменой кода};
      \node[muted,font=\scriptsize,below=0.6em] at (t0.south)
        {при первом вызове};
      \node[muted,font=\scriptsize,below=0.6em] at (t1.south)
        {когда метод стал горячим};
    \end{tikzpicture}
  \end{center}

  \slidegap
  \aside{Первый запрос после старта медленнее второго --- \hi{это не баг}.
    Прогрев --- свойство процесса: перезапустили, и всё начинается заново.}
\end{frame}
\note{Длинный цикл тоже горячий, хотя счётчик вызовов равен единице.}

\begin{frame}{Debug и Release: отличия}
  \slidetop
  \begin{tabular}{@{}p{5.2cm}p{3.0cm}p{3.0cm}@{}}
     & \colhead{accent}{debug} & \colhead{alt}{release} \\[0.8em]
    Оптимизации компилятора   & \hi{нет}        & \lo{да} \\[0.7em]
    Оптимизации JIT           & \hi{запрещены}  & \lo{разрешены} \\[0.7em]
    Файл \texttt{pdb}         & есть            & есть \\[0.7em]
    Назначение                & отладка         & измерение и поставка \\
  \end{tabular}
\end{frame}
\note{В Debug в сборку кладётся прямая пометка для JIT: не оптимизируй тоже.
  pdb есть в обеих -- Release это про оптимизации, а не про символы.}

\begin{frame}[plain]
  \keyline{Ваш код компилируется дважды:\\ в IL на вашей машине, в машинный

\end{frame}
\note{Всё остальное, от прогрева до Release, следует из этого.}

\section{3. Проект и решение}

\begin{frame}[plain]
  \sectionopener{3}{Проект и решение}{csproj --- это манифест для MSBuild,
    а не файл настроек.}
\end{frame}
\note{Бюджет: 15 минут.}

\begin{frame}[fragile]{Файл проекта целиком}
  \slidetop
\begin{lstlisting}[style=xml]
<Project Sdk="Microsoft.NET.Sdk">

  <PropertyGroup>
    <OutputType>Exe</OutputType>
    <TargetFramework>net10.0</TargetFramework>
    <ImplicitUsings>enable</ImplicitUsings>
    <Nullable>enable</Nullable>
  </PropertyGroup>

</Project>
\end{lstlisting}
  \vspace{0.9em}
  \aside{Девять строк, и это всё, что создал \texttt{dotnet new console}.
    Файл короткий потому, что правила лежат в \hi{SDK}, а здесь только
    отклонения от умолчаний.}
\end{frame}
\note{Разбирать строку за строкой. Лет десять назад каждый файл надо было
  перечислять руками.}

\begin{frame}[fragile]{Два способа подключить чужой код}
  \slidetop
\begin{lstlisting}[style=xml]
<ItemGroup>
  <PackageReference Include="Newtonsoft.Json" Version="13.0.3" />
  <ProjectReference Include="../Shared/Shared.csproj" />
</ItemGroup>
\end{lstlisting}
  \vspace{1.1em}
  \versus{Package Reference}{%
    \begin{cbul}{accent}
      \item из реестра NuGet
      \item \hi{замороженная} версия
      \item чтобы получить изменения --- нужен новый пакет
    \end{cbul}}{Project Reference}{%
    \begin{cbul}{alt}
      \item соседний проект в том же репозитории
      \item собирается вместе
      \item \lo{подхватывает} изменения сразу
    \end{cbul}}
\end{frame}
\note{Версия должна быть зафиксирована -- здесь или централизованно. Здесь
  я упрощаю: без Version тоже соберётся.}

\begin{frame}{Решение --- это список, а не контейнер}
  \slidetop
  \claim{Проект живёт сам по себе. Решение просто \hi{перечисляет} проекты.}

  \begin{center}
    \begin{tikzpicture}[font=\small,
      box/.style={draw=hairline,line width=0.8pt,fill=codebg,
                  minimum height=0.85cm,minimum width=2.9cm,align=center,
                  inner sep=0.4em},
      ar/.style={-{Straight Barb[length=1.8mm]},ink,line width=0.7pt}]

\end{frame}
\note{IDE старательно поддерживает недоразумение, что проекты живут внутри

\begin{frame}{Что появляется после сборки}
  \slidetop
  \versus{obj --- черновики}{%
    \begin{cbul}{accent}
      \item \texttt{project.assets.json}
      \item промежуточные файлы компилятора
      \item сгенерированные \texttt{using}
    \end{cbul}}{bin/Debug/net10.0 --- результат}{%
    \begin{cbul}{alt}
      \item готовая dll
      \item всё, что нужно рядом для запуска
      \item подпапка на конфигурацию и версию
    \end{cbul}}

  \slidegap
  \hrulethin{0.42}
  \rulegap
  {\color{alt}\large\bfseries bin и obj не коммитятся в репозиторий.
    Никогда.}

  \slidegap
  \aside{Это производное от исходников: всё получается заново одной командой.}
\end{frame}
\note{obj -- кухня, bin -- окно выдачи. В obj лежат пути с вашей машины,
  и у коллеги они не совпадут.}

% ================================================== 4. DOTNET CLI =========
\section{4. dotnet CLI}

\begin{frame}[plain]
  \sectionopener{4}{dotnet CLI}{Единственный интерфейс к платформе, который
    существует везде, где живёт ваше приложение.}
\end{frame}
\note{Бюджет: 18 минут.}

\begin{frame}{Почему терминал, а не кнопка}
  \slidetop
  \claim{Объяснение не про <<так правильно>>, а про конкретную выгоду.}

  \begin{enumerate}\itemsep0.9em
    \item На сервере сборки, в CI и в контейнере \hi{IDE нет}. Там есть
          ровно эти команды.
    \item IDE --- \hi{надстройка} над ними же. Терминал показывает ошибку
          целиком, а не её сокращённую версию в окошке.
    \item Команду можно \hi{записать и передать}: в файл, в скрипт, в задачу.
          Последовательность кликов --- нельзя.
  \end{enumerate}

  \slidegap
  \aside{За кнопкой <<запустить>> прячется \lo{пять действий}. Вы видите
    одно.}
\end{frame}
\note{Аналогия с механикой и автоматом. LLM, кстати, запускает ровно эти же
  команды -- ничего другого у неё нет.}

\begin{frame}[fragile]{Всё начинается с шаблона}
  \slidetop
\begin{lstlisting}[style=term]
$ dotnet new list

Template Name                            Short Name     Language    Tags
---------------------------------------  -------------  ----------  --------------------
ASP.NET Core Web API                     webapi         [C#],F#     Web/Web API/Service
Blazor Web App                           blazor         [C#]        Web/Blazor
Class Library                            classlib       [C#],F#,VB  Common/Library
Console App                              console        [C#],F#,VB  Common/Console
dotnet gitignore file                    gitignore                  Config
Dotnet local tool manifest file          tool-manifest              Config
global.json file                         globaljson                 Config
MSBuild Directory.Packages.props file    packagesprops              MSBuild/packages/CPM
Solution File                            sln                        Solution
xUnit Test Project                       xunit          [C#],F#,VB  Test/xUnit
\end{lstlisting}
  \vspace{0.7em}
  \aside{Не только проекты: \hi{gitignore}, файл решения и манифест
    инструментов создаются отсюда же. Список сокращён --- шаблонов больше.}
\end{frame}
\note{Показать, что часть шаблонов создаёт один файл, а не проект.}

\begin{frame}{Жизненный цикл проекта}
  \slidetop
  \claim{От рождения до поставки. Запомним \hi{в формате истории}.}

  \begin{center}
    \begin{tikzpicture}[font=\small,x=2.25cm]
      \draw[hairline,line width=0.8pt] (-0.38,0) -- (5.38,0);
      \foreach \i/\c/\k in {0/new/{-n}, 1/restore/{}, 2/build/{-c},
                            3/run/{--project}, 4/test/{}, 5/publish/{-c -o}} {
        \fill[ink] (\i,0) circle (2.2pt);
        \node[above=0.45em,font=\small\ttfamily] at (\i,0) {\c};
        \node[below=0.5em,accent,font=\scriptsize\ttfamily] at (\i,0) {\k};
      }
    \end{tikzpicture}
  \end{center}

  \slidegap
  \aside{Проект рождается, обрастает зависимостями, собирается, запускается,
    проверяется и уезжает на сервер. \hi{\texttt{restore}} --- единственный
    шаг, которому нужна сеть.}
\end{frame}
\note{restore вызывают явно редко: build и run делают его сами.}

\begin{frame}[fragile]{Вся цепочка, три экрана}
  \slidetop
\begin{lstlisting}[style=term]
$ dotnet new console -n HelloWorld
The template "Console App" was created successfully.

$ dotnet run
Hello, World!

$ dotnet run -c Release
Hello, World!

$ dotnet publish -c Release -o ./out
  HelloWorld -> /home/user/HelloWorld/bin/Release/net10.0/HelloWorld.dll
  HelloWorld -> /home/user/HelloWorld/out/

$ ls out
HelloWorld            HelloWorld.dll  HelloWorld.runtimeconfig.json
HelloWorld.deps.json  HelloWorld.pdb
\end{lstlisting}
  \vspace{0.6em}
  \aside{Вторая сборка прошла заново: другая конфигурация --- \hi{другая
    папка} в \texttt{bin}. А в \texttt{out} --- никаких исходников, никакого
    \texttt{obj}, только результат.}
\end{frame}
\note{publish по умолчанию собирает в Release, в отличие от build и run.
  Ключ всё равно пишу явно, чтобы читающий скрипт не гадал.}

\begin{frame}{Инструменты: глобальные и локальные}
  \slidetop
  \claim{Нужен \hi{вам} --- глобальный. Нужен \lo{проекту} --- локальный.}

  \versus{глобальный, ключ -g}{%
    \begin{cbul}{accent}
      \item ставится для вашего пользователя
      \item доступен из любой папки
      \item версии \hi{разъезжаются} между машинами
    \end{cbul}}{локальный, манифест в репозитории}{%
    \begin{cbul}{alt}
      \item перечислен в манифесте вместе с версией
      \item \texttt{dotnet tool restore} после клонирования
      \item у всех \lo{одна и та же} версия
      \item манифест коммитится --- это исходник
    \end{cbul}}
\end{frame}
\note{Я это проходил с инструментом миграций: час на выяснение, что версии
  разные.}

\begin{frame}[plain]
  \keyline{CLI --- не <<сложный способ>>.\\ Это единственный способ, который
    существует везде, где живёт ваше приложение.}
\end{frame}
\note{Сначала команды, потом кнопки. К концу курса кнопки разрешу.}
