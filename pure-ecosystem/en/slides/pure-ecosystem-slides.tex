% =========================================================== 2. PREMISE ====
{\darkslide
\begin{frame}[plain]
  \thoughtbody{What is a program, \hi{actually}?}
\end{frame}}
\note{Chapter 2 -- the premise. Budget: 3.5 minutes.}

\section{The premise}

\begin{frame}
  \bigline{The purpose of a program is to \hi{transform data}.\\[0.5em]
    And the result of a program is also \hi{data}.}
\end{frame}
\note{Everything rests on this. Web service, report generator, trading system,
  or even a string concatenation function. Everything else is machinery.}

\begin{frame}
  \vspace{1.2em}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.46\textwidth}
      {\color{muted}\footnotesize WHAT WE WRITE DOWN}\\[0.7em]
      {\large A sequential set of\\ \textbf{instructions}}\\[1.2em]
      take $\cdot$ do $\cdot$ check $\cdot$ assign $\cdot$ return\\[0.9em]
      \aside{The transformation exists only while the CPU runs. Before that, a
      recipe. After that, gone.}
    \end{column}
    \begin{column}{0.46\textwidth}
      {\color{accent}\footnotesize THE ALTERNATIVE}\\[0.7em]
      {\large A sequential set of\\ \textbf{object states}}\\[1.2em]
      not the steps $\cdot$ the \hi{result itself}\\[0.9em]
      \aside{Already the answer, and simply not yet asked. A program becomes a
      composition.}
    \end{column}
  \end{columns}
\end{frame}
\note{The contrast is the whole premise: instructions that must run, versus
  object states that already exist.}

\begin{frame}
  \vspace{1.4em}
  \banner{Five rules}
  {\large
  \begin{enumerate}
    \setlength{\itemsep}{0.75em}
    \item Only \hi{fields}. No methods.
    \item Logic through \hi{composition of objects}.
    \item \hi{Immutability} --- iron, not by convention.
    \item \hi{Lazy}. Nothing runs except \texttt{new}.
    \item \hi{Deterministic} hash codes.
  \end{enumerate}}
\end{frame}
\note{Rule five looks out of place next to the other four. It is the one
  holding them up.}

\begin{frame}
  \vspace{1.5em}
  \banner{One divergence}
  Elegant Objects: do not expose \hi{fields}.\\[0.5em]
  Pure: almost nothing but read-only fields.

  \vspace{1.3em}
  \hrulethin{0.5}
  \vspace{1.2em}

  {\Large \handstroke{The field is the state}.}

  \vspace{1.2em}
  \aside{Established in exactly one place --- the constructor.}
\end{frame}
\note{A field is not an accessor bolted onto stored state. That is the whole
  design, and the rest of the talk is what falls out of it.}

% ================================================= 3. WHY .NET STOPS YOU ===
{\darkslide
\begin{frame}[plain]
  \thoughtbody{So why not just \hi{write it that way} today?}
\end{frame}}
\note{Chapter 3 -- why .NET stops you. Budget: 6 minutes.}

\section{Why .NET stops you}

\begin{frame}[fragile]{Let us compose two strings}
  \vspace{0.5em}
\begin{lstlisting}
public sealed class String : IComparable, IEnumerable<char>, ...

public readonly struct Int32 : IComparable, IEquatable<int>, ...

public readonly struct Boolean : IComparable, IEquatable<bool>, ...
\end{lstlisting}
  \vspace{1.2em}
  \texttt{sealed}. And every numeric primitive is a \texttt{struct}.

  \vspace{0.8em}
  \aside{You cannot subtype them, decorate them, or write one that computes
  itself when asked. Every primitive in the framework.}
\end{frame}
\note{The door is welded shut, and not just on these three.}

\begin{frame}[fragile]{The real damage is one layer up}
  \vspace{0.8em}
\begin{lstlisting}
string.Concat(string, string)  ->  string
Math.Abs(double)               ->  double
\end{lstlisting}
  \vspace{1.2em}
  {\Large No \hi{seam}. Nothing to plug into.}

  \vspace{1.2em}
  \aside{We reach for LINQ and lambdas because the alternative --- real
  composition --- was structurally removed before we sat down.}
\end{frame}
\note{Every operation in the base class library is written against those
  concrete types. Take this away even if you never install a package.}

\begin{frame}
  \bigline{You cannot fix this with a library.\\[0.7em]
    {\normalsize\normalfont\color{muted}A library sits on top of the problem.
    You have to go underneath it and \handstroke{re-found the primitives}.\par}}
\end{frame}
\note{This is why it is an ecosystem and not an SDK.}

\begin{frame}
  \vfill
  \bignum{9}{interfaces --- and not one method among them.}
  \vfill
\end{frame}
\note{That is the entire foundation. Only fields.}

\begin{frame}[fragile]{The foundation}
  \vspace{0.3em}
\begin{lstlisting}
public interface IBool
{
    public bool BoolValue { get; }
}

public interface INumber<out T>
    where T : System.Numerics.INumber<T>
{
    public T NumberValue { get; }
}

public interface IString : IEnumerable<IChar>
{
    public string TextValue { get; }
}
\end{lstlisting}
  \vspace{0.5em}
  \aside{\texttt{INumber} is covariant --- one abstraction, not fifteen.}
\end{frame}
\note{And IString is an IEnumerable of IChar -- in Pure the characters are
  objects too.}

\begin{frame}
  \vspace{1.5em}
  \banner{Native fields}
  {\Large \texttt{BoolValue}\quad \texttt{NumberValue}\quad \texttt{TextValue}}

  \vspace{1.4em}
  \hrulethin{0.5}
  \vspace{1.2em}

  The \hi{bridge point} --- the one place the ecosystem touches .NET.

  \vspace{1.1em}
  {\large \handstroke{Always computed. Never stored.}}
\end{frame}
\note{TextValue returns a real System.String. That field is not there because I
  wanted it -- it is there because it has to be. It is where the composition
  finally runs.}

\begin{frame}[fragile]{Which makes the date interesting}
  \vspace{0.5em}
\begin{lstlisting}
public interface IDate
{
    public INumber<ushort> Day { get; }
    public INumber<ushort> Month { get; }
    public INumber<ushort> Year { get; }
}
\end{lstlisting}
  \vspace{1.2em}
  {\Large \texttt{IDate} has \hi{no value}.}

  \vspace{1.0em}
  A date is not a thing that \emph{has} a value --- a date \textbf{is} three
  numbers.
\end{frame}
\note{Nothing to unwrap, no bridge back to DateTime. And each of those numbers
  is itself an abstraction that may not have computed yet.}

\begin{frame}[fragile]{What you build on top of it}
  \vspace{0.2em}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
public sealed record Millennium : IDate
{
    public Millennium()
    {
        Day = new UShort(1);
        Month = new UShort(1);
        Year = new UShort(2000);
    }

    public INumber<ushort> Day { get; }
    public INumber<ushort> Month { get; }
    public INumber<ushort> Year { get; }
}
\end{lstlisting}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
IDate birthday = new Millennium();
\end{lstlisting}
  \vspace{0.4em}
  \aside{Named. Reusable. Testable. Fixed at construction, and never anything
  else.}
\end{frame}
\note{Not a DateTime with a magic value in it, and not a constant buried in a
  static helper. If your primitives are opaque, composition stops the moment
  you touch a string. Here it never stops.}

% ======================================================= 4. CONSTRUCTORS ===
{\darkslide
\begin{frame}[plain]
  \thoughtbody{No methods. So where does the \hi{work} go?}
\end{frame}}
\note{Chapter 4 -- the centre of the talk. Budget: 5.5 minutes.}

\section{Constructors}

\begin{frame}
  \vspace{1.5em}
  \banner{The transformation lives in the constructors}
  One constructor \hi{assigns} --- and it turns \hi{private} the moment there is
  a second.\\[0.7em]
  Every other one \hi{converts}: compose objects, then delegate.

  \vspace{1.4em}
  \hrulethin{0.5}
  \vspace{1.2em}

  {\large\ttfamily : this(new Something(...))}

  \vspace{1.1em}
  \aside{That chain \emph{is} the transformation.}
\end{frame}
\note{The naive answer is ``into the fields''. That is the half that misleads
  people. Assignment only -- no work, no validation, no evaluation of results.}

\begin{frame}[fragile]{The simplest component in the ecosystem}
  \vspace{0.3em}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
public sealed record ConcatenatedString : IString
{
    private readonly IEnumerable<IString> _parameters;

    public ConcatenatedString(params IEnumerable<IString> parameters)
    {
        _parameters = parameters;
    }

    public string TextValue => string.Concat(_parameters.Select(x => x.TextValue));
}
\end{lstlisting}
  \vspace{1.0em}
  {\large Not a helper. Not a builder. It \handstroke{is} an \texttt{IString}.}
\end{frame}
\note{It does not produce an IString. The component implements the interface and
  expresses the transformation at the same time -- not two responsibilities,
  one.}

\begin{frame}
  \bigline{Nothing was concatenated.\\[0.7em]
    {\normalsize\normalfont\color{muted}Ten thousand of these is ten thousand
    small objects and \emph{zero} string operations.\par}\vspace{1.1em}
    {\LARGE Nothing runs except \texttt{new}.\par}}
\end{frame}
\note{That is rule four, and it fits in four words.}

\begin{frame}[fragile]{Delegation, one layer up}
  \vspace{0.2em}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
public sealed record WrappedString : IString
{
    private readonly IString _concatenated;

    public WrappedString(IString encloser, IString wrappedValue)
        : this(encloser, wrappedValue, encloser) { }

    public WrappedString(IString prefix, IString wrappedValue, IString suffix)
        : this(new ConcatenatedString(prefix, wrappedValue, suffix)) { }

    private WrappedString(IString concatenated)
    {
        _concatenated = concatenated;
    }

    public string TextValue => _concatenated.TextValue;
}
\end{lstlisting}
  \vspace{0.4em}
  \aside{Nothing is wrapped. A wrapped string \hi{holds a concatenation}.}
\end{frame}
\note{Three constructors, and the only one that touches a field is private. The
  missing suffix is filled with an object, not a value -- even a start index of
  zero is new Zero of ushort, never plain 0.}

\begin{frame}[fragile]{The shape does not change above the primitives}
  \vspace{0.3em}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
public sealed record TotalWithVat : INumber<decimal>
{
    private readonly INumber<decimal> _total;

    public TotalWithVat(INumber<decimal> net, INumber<decimal> rate)
        : this(new Sum<decimal>(net, new Product<decimal>(net, rate))) { }

    private TotalWithVat(INumber<decimal> total)
    {
        _total = total;
    }

    public decimal NumberValue => _total.NumberValue;
}
\end{lstlisting}
  \vspace{0.6em}
  \aside{The business rule is the entire implementation --- \hi{no arithmetic},
  not one operator.}
\end{frame}
\note{Read the public constructor out loud: the total is the sum of the net
  amount and the product of the net amount and the rate. Nowhere for a step to
  hide, because there are no steps.}

\begin{frame}[fragile]{So this is what a program looks like}
  \vspace{0.5em}
\begin{lstlisting}
IString placeholder = new WrappedString(
    new LeftCurlyBracketString(),
    new RandomString(),
    new RightCurlyBracketString()
);

INumber<int> result = new Sum<int>(
    new Difference<int>(new Int(10), new Int(3)),
    new Product<int>(new Int(2), new Int(4))
);
\end{lstlisting}
  \vspace{0.9em}
  \aside{Ordinary values --- pass them, store them, return them, nest them.
  Every move is free.}
\end{frame}
\note{Two statements, and nothing has happened yet. There is nothing inside
  them but references to other objects.}

\begin{frame}{A composition is a tree}
  \vspace{0.4em}
  \begin{center}
    \begin{tikzpicture}[
        font=\ttfamily\footnotesize,
        nd/.style={draw=rule,rounded corners=2pt,inner sep=4pt,fill=white},
        leaf/.style={nd,text=muted},
        level distance=12mm,
        level 1/.style={sibling distance=44mm},
        level 2/.style={sibling distance=22mm},
        edge from parent/.style={draw=rule,thick}
      ]
      \node[nd,text=accent] {Sum<int>}
        child { node[nd] {Difference<int>}
          child { node[leaf] {Int(10)} }
          child { node[leaf] {Int(3)} }
        }
        child { node[nd] {Product<int>}
          child { node[leaf] {Int(2)} }
          child { node[leaf] {Int(4)} }
        };
    \end{tikzpicture}
  \end{center}
  \vspace{0.6em}
  \begin{center}
    \aside{Seven constructors ran. Zero additions, zero multiplications.}
  \end{center}
\end{frame}
\note{The graph is the program, and it is standing still.}

\begin{frame}
  \vspace{1.5em}
  \banner{We name a component for \hi{what it is}}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.45\textwidth}
      {\large
       \texttt{ConcatenatedString}\\[0.45em]
       \texttt{Sum}\\[0.45em]
       \texttt{Difference}\\[0.45em]
       \texttt{CurrentTime}}
    \end{column}
    \begin{column}{0.45\textwidth}
      {\large\color{muted}
       \texttt{StringConcatenator}\\[0.45em]
       \texttt{Calculate}\\[0.45em]
       \texttt{TimeProvider}}
    \end{column}
  \end{columns}

  \vspace{1.7em}
  \aside{Not an agent that performs an action --- a value that stands for one.
  Read a composition out loud and it is a noun phrase.}
\end{frame}
\note{It is a rule, not a habit.}

\begin{frame}
  \vspace{1.8em}
  \banner{So where does it end?}
  {\Large It ends at the \handstroke{native field}.}

  \vspace{1.6em}
  \aside{Read \texttt{.TextValue} --- the graph is walked, every deferred
  operation runs in order, and a concrete .NET value comes out.}
\end{frame}
\note{If nothing computes, when does anything happen? Before that call, your
  entire program is inert.}

\begin{frame}[fragile]{``What if I read it twice?'' --- then it runs twice}
  \vspace{0.1em}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
public sealed record CachedString : IString
{
    private readonly Lazy<string> _lazyValue;

    public CachedString(IString value)
        : this(
            new Lazy<string>(
                () => value.TextValue,
                LazyThreadSafetyMode.ExecutionAndPublication
            )
        )
    { }

    private CachedString(Lazy<string> lazyValue)
    {
        _lazyValue = lazyValue;
    }

    public string TextValue => _lazyValue.Value;
}
\end{lstlisting}
  \vspace{0.2em}
  \aside{Caching is \hi{one more object in the graph}.}
\end{frame}
\note{A composition keeps no result -- it is a description, and descriptions do
  not remember. Wrapped, the graph is walked exactly once, thread-safely,
  however many reads. And you decide where in the graph it sits.}

% ==================================================== 5. DETERMINED HASH ===
{\darkslide
\begin{frame}[plain]
  \thoughtbody{But what is \hi{identity}?}
\end{frame}}
\note{Chapter 5 -- determined hashes. Budget: 5.5 minutes.}

\section{Determined hashes}

\begin{frame}
  \bigline{\handstroke{Determined hashes}.\\[0.7em]
    {\normalsize\normalfont\color{muted}Not \texttt{GetHashCode}. Not
    \texttt{Equals}. At all.\par}}
\end{frame}
\note{Principle five. The one that looked out of place next to the other four.}

\begin{frame}
  \vspace{1.5em}
  \banner{\texttt{GetHashCode} is not identity}
  {\Large A \hi{different answer} the next time you run the same program.}

  \vspace{1.5em}
  Cannot go in a file. Cannot go over a network. Cannot be compared to
  yesterday.

  \vspace{1.4em}
  \aside{What we want is the hash of an object \hi{state snapshot} --- the same
  answer tomorrow, on another machine, in another runtime.}
\end{frame}
\note{Deliberate security measure since .NET Core -- not a bug. But the
  framework's notion of sameness has a lifetime shorter than the
  application's.}

\begin{frame}
  \vspace{1.3em}
  \begin{center}
    \Large
    \texttt{0.GetHashCode()} \quad $=$ \quad \texttt{0}\\[0.65em]
    \texttt{false.GetHashCode()} \quad $=$ \quad \texttt{0}\\[0.65em]
    \texttt{default(char).GetHashCode()} \quad $=$ \quad \texttt{0}
  \end{center}
  \vspace{1.6em}
  \begin{center}
    Three different values, from three different types,
    \handstroke{sharing one identity}.
  \end{center}
\end{frame}
\note{And this second problem is worse than the first. Pause on the three
  zeroes.}

\begin{frame}[fragile]{What we use instead}
  \vspace{0.5em}
\begin{lstlisting}
public interface IDeterminedHash : IEnumerable<byte>;
\end{lstlisting}
  \vspace{0.8em}
  An identity is not an integer --- it is a \hi{sequence of bytes}.

  \vspace{1.2em}
\begin{lstlisting}
public IEnumerator<byte> GetEnumerator()
{
    return SHA256.HashData([.. _bytes]).AsEnumerable().GetEnumerator();
}
\end{lstlisting}
  \vspace{0.8em}
  \aside{SHA-256. Thirty-two bytes. Deterministic across processes, machines and
  runtimes --- which is the entire point.}
\end{frame}
\note{The interface is the entire file. And it is lazy, like everything else:
  enumerate it and it computes; leave it alone and it costs nothing.}

\begin{frame}[fragile]{A domain separator per type}
  \vspace{0.7em}
\begin{lstlisting}
private static readonly byte[] TypePrefix =
[
    0, 69, 151, 1, 4, 52, 46, 126, 159, 32, 211, 174, 149, 230, 168, 150,
];
\end{lstlisting}
  \vspace{1.3em}
  Sixteen random bytes, unique per type, prepended before hashing.

  \vspace{1.1em}
  \aside{So zero, false and the empty string give three completely different
  digests. The collision the framework hands you for free is structurally
  impossible.}
\end{frame}
\note{Every type gets one.}

\begin{frame}
  \vspace{1.5em}
  \banner{No \texttt{GetHashCode} means no framework collections}
  A \texttt{Dictionary} asks its keys for a hash code. \hi{Ours do not answer.}

  \vspace{1.4em}
  \hrulethin{0.5}
  \vspace{1.2em}

  So the ecosystem ships its own, keyed on determined hashes.
\end{frame}
\note{Two keys are the same when their byte sequences are the same.}

\begin{frame}
  \vfill
  \bignum{256}{bits decide equality. The \texttt{int} is only a
    \hi{bucket index} now.}
  \vfill
\end{frame}
\note{The integer has not disappeared -- it is demoted from an identity to a
  bucket index. It only answers ``roughly where should I look'', and collisions
  there are harmless, because the real comparison is exact.}

\begin{frame}
  \vspace{1.8em}
  \banner{One door this opens}
  {\Large Has an object with this state \hi{already been allocated}?}

  \vspace{1.6em}
  If yes --- there is no need to allocate again. Return the existing reference.

  \vspace{1.4em}
  \aside{Not built. But the door is open, and only because of immutability plus
  determined state.}
\end{frame}
\note{Mark it as an opportunity, not a claim. Then move on.}

% ======================================================= 6. CONTROL FLOW ===
{\darkslide
\begin{frame}[plain]
  \thoughtbody{And what about \hi{\texttt{if}}?}
\end{frame}}
\note{Chapter 6 -- control flow. Budget: 5 minutes. You cannot compose a
  keyword, and a language without branching is not a language.}

\section{Control flow}

\begin{frame}[fragile]{An \texttt{if} that produces a string \emph{is} a string}
  \vspace{0.1em}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
public sealed record StringChoice : IString
{
    private readonly IBool _condition;
    private readonly IString _valueOnTrue;
    private readonly IString _valueOnFalse;

    public StringChoice(IBool condition, IString valueOnTrue, IString valueOnFalse)
    {
        _condition = condition;
        _valueOnTrue = valueOnTrue;
        _valueOnFalse = valueOnFalse;
    }

    public string TextValue =>
        _condition.BoolValue ? _valueOnTrue.TextValue : _valueOnFalse.TextValue;
}
\end{lstlisting}
  \vspace{0.3em}
  \aside{The condition is an object. Both branches are objects. And the choice
  \hi{implements the target interface}.}
\end{frame}
\note{It is not a control structure that yields a value. It is a value that
  happens to have a decision inside it.}

\begin{frame}[fragile]{Which means it composes}
  \vspace{0.5em}
\begin{lstlisting}
INumber<int> count = new Int(3);

IString label = new WrappedString(
    new LeftCurlyBracketString(),
    new StringChoice(
        new GreaterThanCondition<int>(count, new Zero<int>()),
        new String(count),
        new EmptyString()
    ),
    new RightCurlyBracketString()
);
\end{lstlisting}
  \vspace{0.9em}
  \aside{The \texttt{WrappedString} has no idea there is a branch inside it. It
  was handed an \texttt{IString}.}
\end{frame}
\note{A choice inside a concatenation, inside another choice, inside a cache --
  nothing downstream needs to know a decision is in there.}

\begin{frame}[fragile]{When the interface has more than one field}
  \vspace{0.1em}
\begin{lstlisting}[basicstyle=\ttfamily\tiny]
public sealed record DateChoice : IDate
{
    public DateChoice(IBool condition, IDate valueOnTrue, IDate valueOnFalse)
        : this(
            new NumberChoice<ushort>(condition, valueOnTrue.Day, valueOnFalse.Day),
            new NumberChoice<ushort>(condition, valueOnTrue.Month, valueOnFalse.Month),
            new NumberChoice<ushort>(condition, valueOnTrue.Year, valueOnFalse.Year)
        )
    { }

    private DateChoice(INumber<ushort> day, INumber<ushort> month, INumber<ushort> year)
    {
        Day = day;
        Month = month;
        Year = year;
    }

    public INumber<ushort> Day { get; }
    public INumber<ushort> Month { get; }
    public INumber<ushort> Year { get; }
}
\end{lstlisting}
  \vspace{0.3em}
  \aside{The public constructor performs no branching at all --- it composes
  \hi{three smaller choices}, one per field, and delegates.}
\end{frame}
\note{The constructor pattern from the previous chapter, now applied to
  branching. The private one assigns.}

\begin{frame}
  \vspace{1.6em}
  \banner{The \texttt{if} is distributed}
  {\Large There is \hi{no single branch point}.}

  \vspace{1.5em}
  Every field decides on its own, at the moment it is read.

  \vspace{1.4em}
  \aside{Ask only for the year and the day is never chosen --- that branch never
  happens at all. A statement cannot do that.}
\end{frame}
\note{A statement executes once, in one place, at one time. This is a decision
  that exists in three places and may partially never occur.}
